local spell = Spell("instant")

-- Create area OUTSIDE the function (during script loading)

    local area = createCombatArea({
	{1, 1, 1},
	{1, 3, 1},
	{1, 1, 1},
    })
	

local combat = Combat()
combat:setParameter(COMBAT_PARAM_TYPE, COMBAT_PHYSICALDAMAGE)
combat:setParameter(COMBAT_PARAM_EFFECT, CONST_ME_HITAREA)
combat:setParameter(COMBAT_PARAM_USECHARGES, true)
combat:setArea(area)
combat:setFormula(COMBAT_FORMULA_LEVELMAGIC, -3.5, 0, -5.4, 0)

function spell.onCastSpell(creature, variant)
    local player = creature:getPlayer()
    if not player then
        return false
    end

    -- Check promotion
    local vocation = player:getVocation()
    local vocationId = vocation:getId()
    if vocationId ~= 4 and vocationId ~= 13 then
        player:sendCancelMessage("Only Deadly Wizards can use this spell.")
        player:getPosition():sendMagicEffect(CONST_ME_POFF)
        return false
    end

    -- Check mana
    if player:getMana() < 400 then
        player:sendCancelMessage("You do not have enough mana.")
        player:getPosition():sendMagicEffect(CONST_ME_POFF)
        return false
    end

    -- Consume mana
    player:addMana(-400)
    player:addManaSpent(400)
    
    -- Execute the pre-configured combat
    return combat:execute(creature, variant)
end

spell:name("Deadly Berseker")
spell:words("exevo exori")
spell:group("attack")
spell:vocation("knight;true", "elite knight;true")
spell:id(4) -- Assign unique ID
spell:cooldown(2 * 1000)
spell:groupCooldown(2 * 1000)
spell:level(100)
spell:mana(400)
spell:isPremium(true)
spell:needTarget(false)
spell:isAggressive(true)
spell:needWeapon(true)
spell:register()
